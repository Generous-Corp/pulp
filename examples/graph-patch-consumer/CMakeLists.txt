cmake_minimum_required(VERSION 3.24)
project(GraphPatchConsumer LANGUAGES CXX)
find_package(Pulp CONFIG REQUIRED COMPONENTS host)
add_executable(graph-patch-consumer main.cpp)
target_link_libraries(graph-patch-consumer PRIVATE Pulp::host)
target_compile_features(graph-patch-consumer PRIVATE cxx_std_20)
enable_testing()
add_test(NAME graph-patch-hello COMMAND graph-patch-consumer "${CMAKE_CURRENT_SOURCE_DIR}/hello.pat")
set_tests_properties(graph-patch-hello PROPERTIES TIMEOUT 30)
